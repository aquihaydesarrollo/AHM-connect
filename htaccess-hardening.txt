# FUERA de los marcadores "# BEGIN WordPress" / "# END WordPress" (si no,
# WordPress lo borra al regenerar los enlaces permanentes).

# --- Bloquear ficheros que filtran versión / son sensibles ---
<FilesMatch "^(readme\.html|license\.txt|wp-config\.php|wp-config-sample\.php|\.htaccess|\.env|.*\.(bak|old|sql|log|zip))$">
    Require all denied
</FilesMatch>

# --- Bloquear xmlrpc.php (evita amplificación / brute-force) ---
<Files "xmlrpc.php">
    Require all denied
</Files>

# --- Cabeceras de seguridad (cubre también ficheros estáticos) ---
#
# OJO CON EL FRAMING. Si el sitio se integra con un portal de compras externo
# (PunchOut cXML/OCI: SAP Ariba, Coupa, Jaggaer…), ese portal abre el catálogo
# DENTRO DE UN IFRAME y "X-Frame-Options: SAMEORIGIN" se lo impide: el navegador
# descarga la página — 200 en el log — pero se niega a pintarla, y el comprador
# ve una pantalla en blanco. X-Frame-Options no admite lista blanca (ALLOW-FROM
# está obsoleto), así que en esos sitios hay que SUSTITUIRLO por la CSP de abajo,
# nunca poner las dos: algunos navegadores priorizan X-Frame-Options.
#
#     Header always unset X-Frame-Options
#     Header always set Content-Security-Policy "frame-ancestors 'self' https://*.ariba.com"
#
<IfModule mod_headers.c>
    Header always set X-Content-Type-Options "nosniff"
    Header always set X-Frame-Options "SAMEORIGIN"
    Header always set Content-Security-Policy "frame-ancestors 'self'"
    Header always set Referrer-Policy "strict-origin-when-cross-origin"
    Header always set Permissions-Policy "geolocation=(), microphone=(), camera=()"
    Header always set Strict-Transport-Security "max-age=31536000; includeSubDomains; preload" env=HTTPS
    Header always unset X-Powered-By
    Header unset X-Powered-By
</IfModule>

# --- Ocultar versión PHP a nivel servidor (si LiteSpeed lo permite) ---
<IfModule mod_php.c>
    php_flag expose_php off
</IfModule>

# --- Caché de estáticos (arregla "cache TTL: 0" de Lighthouse) ---
# Ahorra ~1.5MB en visitas repetidas y acelera el LCP.
<IfModule mod_expires.c>
    ExpiresActive On
    ExpiresByType text/css                "access plus 1 year"
    ExpiresByType application/javascript  "access plus 1 year"
    ExpiresByType text/javascript         "access plus 1 year"
    ExpiresByType image/webp              "access plus 1 year"
    ExpiresByType image/png               "access plus 1 year"
    ExpiresByType image/jpeg              "access plus 1 year"
    ExpiresByType image/gif               "access plus 1 year"
    ExpiresByType image/svg+xml           "access plus 1 year"
    ExpiresByType font/woff2              "access plus 1 year"
    ExpiresByType application/font-woff2  "access plus 1 year"
</IfModule>
<IfModule mod_headers.c>
    <FilesMatch "\.(css|js|webp|png|jpe?g|gif|svg|woff2)$">
        Header set Cache-Control "public, max-age=31536000, immutable"
    </FilesMatch>
</IfModule>
